const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); // ========================================== // CONFIGURAÇÕES DO JOGO (SERVIDOR) // ========================================== const TICK_RATE = 60; const MAP_SIZE = 2500; const PLAYER_SPEED = 5; const PLAYER_RADIUS = 20; const BULLET_SPEED = 15; const MAX_PLAYERS = 20; const KILLS_TO_WIN = 10; const rooms = {}; // Gera um mapa procedural function generateMap() { const map = { walls: [], trees: [], rocks: [] }; // Bordas do mapa map.walls.push({x: -50, y: -50, w: MAP_SIZE+100, h: 50}); // Topo map.walls.push({x: -50, y: MAP_SIZE, w: MAP_SIZE+100, h: 50}); // Base map.walls.push({x: -50, y: -50, w: 50, h: MAP_SIZE+100}); // Esq map.walls.push({x: MAP_SIZE, y: -50, w: 50, h: MAP_SIZE+100}); // Dir // Casas (Muros) for(let i=0; i<15; i++) { map.walls.push({ x: Math.random() * (MAP_SIZE - 200) + 100, y: Math.random() * (MAP_SIZE - 200) + 100, w: Math.random() * 150 + 50, h: Math.random() * 150 + 50 }); } // Árvores for(let i=0; i<40; i++) { map.trees.push({ x: Math.random() * MAP_SIZE, y: Math.random() * MAP_SIZE, r: Math.random() * 30 + 20 }); } // Pedras for(let i=0; i<30; i++) { map.rocks.push({ x: Math.random() * MAP_SIZE, y: Math.random() * MAP_SIZE, r: Math.random() * 20 + 10 }); } return map; } // Lógica de colisão Círculo vs Retângulo function checkCircleRect(circle, rect) { let testX = circle.x; let testY = circle.y; if (circle.x < rect.x) testX = rect.x; else if (circle.x > rect.x + rect.w) testX = rect.x + rect.w; if (circle.y < rect.y) testY = rect.y; else if (circle.y > rect.y + rect.h) testY = rect.y + rect.h; let distX = circle.x - testX; let distY = circle.y - testY; return (distX*distX + distY*distY) <= (circle.r*circle.r); } // Lógica Círculo vs Círculo function checkCircleCircle(c1, c2) { let dx = c1.x - c2.x; let dy = c1.y - c2.y; return (dx*dx + dy*dy) <= ((c1.r + c2.r) * (c1.r + c2.r)); } io.on('connection', (socket) => { socket.on('joinRoom', ({ name, roomId }) => { if (!rooms[roomId]) { rooms[roomId] = { map: generateMap(), players: {}, bullets: [], over: false }; } if (Object.keys(rooms[roomId].players).length >= MAX_PLAYERS) { socket.emit('error', 'Sala cheia!'); return; } socket.join(roomId); socket.roomId = roomId; rooms[roomId].players[socket.id] = { id: socket.id, name: name.substring(0,10), x: Math.random() * (MAP_SIZE-100) + 50, y: Math.random() * (MAP_SIZE-100) + 50, r: PLAYER_RADIUS, angle: 0, hp: 100, kills: 0, dead: false }; socket.emit('init', { map: rooms[roomId].map, id: socket.id, MAP_SIZE }); io.to(roomId).emit('chat', { sender: 'Sistema', msg: `${name} entrou na partida!` }); }); socket.on('input', (data) => { const room = rooms[socket.roomId]; if(!room || !room.players[socket.id] || room.players[socket.id].dead || room.over) return; let p = room.players[socket.id]; let oldX = p.x; let oldY = p.y; p.angle = data.angle; p.x += data.dx * PLAYER_SPEED; p.y += data.dy * PLAYER_SPEED; // Colisões com o mapa (Paredes) room.map.walls.forEach(w => { if (checkCircleRect(p, w)) { p.x = oldX; p.y = oldY; } }); room.map.rocks.forEach(rock => { if (checkCircleCircle(p, rock)) { p.x = oldX; p.y = oldY; } }); // Tiro if (data.fire && (!p.lastFire || Date.now() - p.lastFire > 200)) { p.lastFire = Date.now(); room.bullets.push({ id: Math.random(), ownerId: p.id, x: p.x + Math.cos(p.angle) * 30, y: p.y + Math.sin(p.angle) * 30, vx: Math.cos(p.angle) * BULLET_SPEED, vy: Math.sin(p.angle) * BULLET_SPEED, life: 60, r: 5 }); io.to(socket.roomId).emit('sound', 'shoot'); } }); socket.on('chat', (msg) => { const room = rooms[socket.roomId]; if(room && room.players[socket.id]) { io.to(socket.roomId).emit('chat', { sender: room.players[socket.id].name, msg: msg.substring(0, 50) }); } }); socket.on('disconnect', () => { const room = rooms[socket.roomId]; if (room && room.players[socket.id]) { io.to(socket.roomId).emit('chat', { sender: 'Sistema', msg: `${room.players[socket.id].name} saiu.` }); delete room.players[socket.id]; if (Object.keys(room.players).length === 0) delete rooms[socket.roomId]; } }); }); // Loop de Física do Servidor setInterval(() => { for (let roomId in rooms) { let room = rooms[roomId]; if(room.over) continue; // Move Bullets for (let i = room.bullets.length - 1; i >= 0; i--) { let b = room.bullets[i]; b.x += b.vx; b.y += b.vy; b.life--; let hit = false; // Colisão Tiro vs Parede room.map.walls.forEach(w => { if(checkCircleRect(b, w)) hit = true; }); room.map.rocks.forEach(r => { if(checkCircleCircle(b, r)) hit = true; }); // Colisão Tiro vs Player for (let pid in room.players) { let p = room.players[pid]; if (!p.dead && pid !== b.ownerId && checkCircleCircle(b, p)) { hit = true; p.hp -= 20; io.to(roomId).emit('sound', 'hit'); io.to(roomId).emit('particle', {x: p.x, y: p.y, color: '#f00'}); if (p.hp <= 0) { p.dead = true; p.hp = 0; if(room.players[b.ownerId]) room.players[b.ownerId].kills++; io.to(roomId).emit('sound', 'die'); io.to(roomId).emit('chat', { sender: 'Sistema', msg: `${p.name} foi eliminado por ${room.players[b.ownerId].name}` }); // Checa Vitória if (room.players[b.ownerId].kills >= KILLS_TO_WIN) { room.over = true; io.to(roomId).emit('victory', room.players[b.ownerId].name); setTimeout(() => { delete rooms[roomId]; io.to(roomId).disconnectSockets(); }, 5000); } else { // Respawn setTimeout(() => { if(room.players[pid]) { room.players[pid].x = Math.random() * (MAP_SIZE-100) + 50; room.players[pid].y = Math.random() * (MAP_SIZE-100) + 50; room.players[pid].hp = 100; room.players[pid].dead = false; } }, 3000); } } } } if (hit || b.life <= 0) { if(hit) io.to(roomId).emit('particle', {x: b.x, y: b.y, color: '#ff0'}); room.bullets.splice(i, 1); } } // Envia estado atual para os jogadores io.to(roomId).emit('update', { players: room.players, bullets: room.bullets }); } }, 1000 / TICK_RATE); // ========================================== // FRONT-END HTML + CSS + JS EMBARCADO // ========================================== const HTML_CONTENT = ` Canvas Shooter MMO

A sala será reiniciada em breve...

Ranking (Kills)

`; // Servir a aplicação enviando a string HTML diretamente (Sem precisar criar arquivo) app.get('/', (req, res) => { res.send(HTML_CONTENT); }); // Inicialização const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(\`✅ BATTLE CANVAS LIGADO!\`); console.log(\`Jogue abrindo o navegador em: http://localhost:\${PORT}\`); console.log(\`Para jogar pelo celular, conecte no mesmo Wi-Fi e acesse o IP do seu PC na porta \${PORT}.\`); });
